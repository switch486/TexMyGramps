      "type": "string"
    },
    "personPicture_mediaObjectID": {
      "type": "string"
    },
    "detailPages": {

        "type": "object",

            "type": "string"

            "type": "string"
          }

        "required": ["mediaObjectID"],

      }

  "required": ["personGrampsID"],
  "additionalProperties": false
}
